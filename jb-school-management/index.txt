  <script>
    // ====== CHECK IF RUNNING IN ELECTRON ======
    const isElectron = typeof window !== 'undefined' && window.electronAPI;

    // ====== DATABASE CONNECTION ======
    async function connectToDatabase() {
      const statusDiv = document.getElementById('dbStatus');
      
      if (isElectron) {
        // Running in Electron - real database connection
        try {
          statusDiv.textContent = 'Opening file browser...';
          statusDiv.className = 'db-status';
          statusDiv.style.display = 'block';
          
          // Select database file
          const filePath = await window.electronAPI.selectDatabase();
          
          if (!filePath) {
            statusDiv.textContent = 'No file selected. Please try again.';
            statusDiv.className = 'db-status error';
            return;
          }
          
          statusDiv.textContent = 'Connecting to database...';
          
          // Connect to database
          const result = await window.electronAPI.connectDatabase(filePath);
          
          if (result.success) {
            statusDiv.textContent = '✅ Connected successfully!';
            statusDiv.className = 'db-status success';
            
            // Load data from database
            await loadAllData();
            
            // Hide overlay after 1 second
            setTimeout(() => {
              document.getElementById('dbConnectOverlay').style.display = 'none';
            }, 1000);
          } else {
            statusDiv.textContent = '❌ Connection failed: ' + result.message;
            statusDiv.className = 'db-status error';
          }
        } catch (error) {
          statusDiv.textContent = '❌ Error: ' + error.message;
          statusDiv.className = 'db-status error';
        }
      } else {
        // Running in web browser - simulated connection
        statusDiv.textContent = 'Simulating database connection...';
        statusDiv.className = 'db-status';
        statusDiv.style.display = 'block';
        
        setTimeout(() => {
          statusDiv.textContent = '✅ Demo mode - using sample data';
          statusDiv.className = 'db-status success';
          
          setTimeout(() => {
            document.getElementById('dbConnectOverlay').style.display = 'none';
          }, 1500);
        }, 1000);
      }
    }

    // ====== LOAD ALL DATA FROM DATABASE ======

        if (teachersResult.success) {
          teachers = teachersResult.data;
          refreshTeacherTable();
        }
        
        // Load subjects
        const subjectsResult = await window.electronAPI.getSubjects();
        if (subjectsResult.success) {
          subjects = subjectsResult.data;
          refreshSubjectTable();
        }
        
        // Load classes
        const classesResult = await window.electronAPI.getClasses();
        if (classesResult.success) {
          classes = classesResult.data;
        }
        
        // Load assessments
        const assessmentsResult = await window.electronAPI.getAssessments();
        if (assessmentsResult.success) {
          assessments = assessmentsResult.data;
          refreshAssessmentTable();
        }
        
        // Load fees
        const feesResult = await window.electronAPI.getFees();
        if (feesResult.success) {
          fees = feesResult.data;
          refreshFeesTable();
        }
        
        // Load attendance
        const attendanceResult = await window.electronAPI.getAttendance();
        if (attendanceResult.success) {
          attendance = attendanceResult.data;
          refreshAttendanceTable();
        }
        
        // Load school settings
        const settingsResult = await window.electronAPI.getSchoolSettings();
        if (settingsResult.success && settingsResult.data) {
          schoolSettings = {
            schoolName: settingsResult.data.SchoolName || 'JB School Management System',
            address: settingsResult.data.Address || '',
            phone: settingsResult.data.Phone || '',
            email: settingsResult.data.Email || '',
            website: settingsResult.data.Website || '',
            motto: settingsResult.data.Motto || '',
            logoPath: settingsResult.data.LogoPath || ''
          };
        }
        
        // Update UI
        populateClassDropdowns();
        populateStudentDropdowns();
        populateSubjectDropdowns();
        populateAssessmentClassDropdown();
        loadSchoolSettings();
        updateDashboard();
        
      } catch (error) {
        console.error('Error loading data:', error);
        alert('Error loading data from database: ' + error.message);
      }
    }

    function searchTable(tableId, query) {
      const table = document.getElementById(tableId);
      const rows = table.getElementsByTagName('tr');
      
      for (let i = 1; i < rows.length; i++) {
        const row = rows[i];
        const text = row.textContent.toLowerCase();
        row.style.display = text.includes(query.toLowerCase()) ? '' : 'none';
      }
    }

    // ====== GRADE CALCULATION (Ghana WAEC System) ======
    function calculateGrade(total) {
      if (total >= 80) return '1 - Distinction';
      if (total >= 70) return '2 - Credit';
      if (total >= 60) return '3 - Credit';
      if (total >= 50) return '4 - Pass';
      if (total >= 40) return '5 - Pass';
      if (total >= 35) return '6 - Pass';
      return '7 - Fail';
    }

    // ====== GENERATE CLASS LIST ======
    function generateClassList() {
      const selectedClassId = parseInt(document.getElementById('classListSelect').value);
      const display = document.getElementById('classListDisplay');
      
      if (!selectedClassId) {
        display.innerHTML = '';
        return;
      }
      
      const selectedClass = classes.find(c => c.ClassID === selectedClassId);
      const classStudents = students.filter(s => s.ClassID === selectedClassId);
      
      // Sort: Girls first, then boys, both alphabetically
      const girls = classStudents.filter(s => s.Gender === 'F').sort((a, b) => a.FirstName.localeCompare(b.FirstName));
      const boys = classStudents.filter(s => s.Gender === 'M').sort((a, b) => a.FirstName.localeCompare(b.FirstName));
      const sortedStudents = [...girls, ...boys];
      
      if (sortedStudents.length === 0) {
        display.innerHTML = '<div class="card"><p>No students found in this class.</p></div>';
        return;
      }
      
      let html = `
        <div class="card" style="margin-top: 20px;">
          <div class="print-only">
            <h2 style="text-align: center; color: #2e8b57;">Greenfield International School</h2>
            <h3 style="text-align: center;">Class List - ${selectedClass.ClassName}</h3>
            <p style="text-align: center;">Total Students: ${sortedStudents.length}</p>
            <hr>
          </div>
          <div class="table-container">
            <table>
              <tr>
                <th>No.</th>
                <th>Admission No</th>
                <th>Full Name</th>
                <th>Gender</th>
                <th>DOB</th>
                <th>Admission Date</th>
                <th>Parent Name</th>
                <th>Contact</th>
                <th>Address</th>
              </tr>
      `;
      
      sortedStudents.forEach((student, index) => {
        html += `
          <tr>
            <td>${index + 1}</td>
            <td>${student.AdmissionNo}</td>
            <td>${student.FirstName} ${student.LastName}</td>
            <td>${student.Gender === 'F' ? 'Female' : 'Male'}</td>
            <td>${student.DOB}</td>
            <td>${student.AdmissionDate}</td>
            <td>${student.ParentName}</td>
            <td>${student.Contact}</td>
            <td>${student.Address}</td>
          </tr>
        `;
      });
      
      html += `
            </table>
          </div>
        </div>
      `;
      
      display.innerHTML = html;
    }

    function updateArrearsTable() {
      const arrearsTable = document.getElementById('arrearsTable');

            <td style="color: #f44336; font-weight: bold;">₵${fee.Balance.toFixed(2)}</td>

        StudentID: studentId,
        Term: term,

        AmountBilled: billed,
        AmountPaid: paid,
